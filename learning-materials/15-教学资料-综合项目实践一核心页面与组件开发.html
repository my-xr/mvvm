三、核心页面代码

3.1 登录页 Login.vue（K15-2）

<template>
  <form @submit.prevent="onLogin">
    <input v-model.trim="form.username" placeholder="用户名" />
    <input v-model.trim="form.password" type="password" placeholder="密码" />
    <button type="submit">登录</button>
    <p v-if="error" class="err">{{ error }}</p>
  </form>
</template>
<script setup>
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { useUserStore } from '../stores/user'
const form = reactive({ username: '', password: '' })
const error = ref('')
const router = useRouter()
const userStore = useUserStore()
const onLogin = async () => {
  if (!form.username || form.password.length < 6) {
    error.value = '用户名不能空且密码至少6位'
    return
  }
  try {
    await userStore.login(form)
    router.push('/projects')
  } catch (e) {
    error.value = '登录失败：' + e.message
  }
}
</script>
<style scoped>
.err { color: red; font-size: 12px; }
</style>

3.3 项目列表页 Projects.vue（K15-3）

<template>
  <div>
    <h2>项目列表</h2>
    <input v-model="keyword" placeholder="搜索" />
    <ul v-if="loading === false">
      <li v-for="p in filtered" :key="p.id" @click="goTasks(p.id)">
        {{ p.title }} - {{ p.status }}
      </li>
    </ul>
    <p v-else>加载中...</p>
  </div>
</template>
<script setup>
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { getProjects } from '../api/project'
const router = useRouter()
const keyword = ref('')
const projects = ref([])
const loading = ref(true)
const filtered = computed(() => projects.value.filter(p => p.title.includes(keyword.value)))
const goTasks = (id) => router.push(`/projects/${id}/tasks`)
onMounted(async () => {
  try {
    projects.value = await getProjects()
  } finally {
    loading.value = false
  }
})
</script>

<template>
  <div>
    <h2>{{ task.title }}</h2>
    <p>状态：{{ task.status }}</p>
    <p>截止：{{ task.deadline }}</p>
    <button @click="goSubmit">提交成果</button>
  </div>
</template>
<script setup>
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { getTaskById } from '../api/task'
const props = defineProps(['id'])
const router = useRouter()
const task = ref({})
const goSubmit = () => router.push(`/tasks/${props.id}/submit`)
onMounted(async () => {
  task.value = await getTaskById(props.id)
})
</script>

3.5 成果提交页 SubmitForm.vue（K15-4）

<template>
  <form @submit.prevent="onSubmit">
    <textarea v-model="form.content" placeholder="成果内容"></textarea>
    <button type="submit">提交</button>
  </form>
</template>
<script setup>
import { reactive } from 'vue'
import { useRouter } from 'vue-router'
import { submitTask } from '../api/task'
const props = defineProps(['id'])
const form = reactive({ content: '' })
const router = useRouter()
const onSubmit = async () => {
  await submitTask(props.id, form)
  alert('提交成功')
  router.back()
}
</script>

3.6 评审页 ReviewDetail.vue（K15-4）

<template>
  <div>
    <h2>评审</h2>
    <p>提交内容：{{ submission.content }}</p>
    <input v-model.number="score" type="number" placeholder="分数" />
    <textarea v-model="comment" placeholder="评语"></textarea>
    <button @click="onReview">提交评审</button>
  </div>
</template>
<script setup>
import { ref, onMounted } from 'vue'
import { getSubmission, reviewSubmission } from '../api/review'
const props = defineProps(['id'])
const submission = ref({})
const score = ref(0)
const comment = ref('')
onMounted(async () => { submission.value = await getSubmission(props.id) })
const onReview = async () => {
  await reviewSubmission(props.id, { score: score.value, comment: comment.value })
  alert('评审完成')
}
</script>

四、接口模块（src/api/）

// api/project.js
import request from '../utils/request'
export const getProjects = () => request.get('/projects')
// api/task.js
import request from '../utils/request'
export const getTasks = (projectId) => request.get(`/projects/${projectId}/tasks`)
export const getTaskById = (id) => request.get(`/tasks/${id}`)
export const submitTask = (id, data) => request.post(`/tasks/${id}/submit`, data)
// api/review.js
import request from '../utils/request'
export const getReviews = () => request.get('/reviews')
export const getSubmission = (id) => request.get(`/reviews/${id}`)
export const reviewSubmission = (id, data) => request.post(`/reviews/${id}`, data)
// api/user.js
import request from '../utils/request'
export const login = (data) => request.post('/login', data)
export const getUserInfo = () => request.get('/user/info')
export const getUsers = () => request.get('/admin/users')
