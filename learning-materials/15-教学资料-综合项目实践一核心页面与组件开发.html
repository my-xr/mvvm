本课实践任务与验收

请把下面任务按顺序完成。每一步都要在浏览器中验证，不能只复制代码而不检查页面行为。

  1. 创建 Vite + Vue 3 项目，安装 vue-router、pinia 和 axios，按目录约定创建文件。
  2. 先将 VITE_USE_MOCK=true 写入 .env.local，完成登录、项目、任务、成果提交、评审、用户管理和个人中心页面。
  3. 分别使用学生、教师、管理员账号测试菜单显示与路由权限：学生提交成果，教师评审，管理员管理用户。
  4. 主动测试加载中、接口失败、空列表、任务不存在和评审记录不存在等状态，确保页面有明确提示且不会提交无效数据。
  5. 最后运行 npm run build，保存构建成功截图和关键页面截图；第 16 课再把项目列表本地状态迁移到 projectStore 并切换真实接口。

验收标准：刷新受保护页面会回到登录页；角色不符不能进入页面；空数据和错误状态可见；表单校验有效；构建无报错。

3.1 登录页 Login.vue（K15-2）

<template>
  <form @submit.prevent="onLogin">
    <input v-model.trim="form.username" placeholder="用户名" />
    <input v-model.trim="form.password" type="password" placeholder="密码" />
    <select v-model="form.role">
      <option value="student">学生</option>
      <option value="teacher">教师</option>
      <option value="admin">管理员</option>
    </select>
    <button type="submit">登录</button>
    <p v-if="error" class="err">{{ error }}</p>
  </form>
</template>
<script setup>
import { reactive, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useUserStore } from '../stores/user'
const form = reactive({ username: '', password: '', role: 'student' })
const error = ref('')
const route = useRoute()
const router = useRouter()
const userStore = useUserStore()
const onLogin = async () => {
  if (!form.username || form.password.length < 6) {
    error.value = '用户名不能空且密码至少6位'
    return
  }
  try {
    await userStore.login(form)
    router.replace(String(route.query.redirect || '/projects'))
  } catch (e) {
    error.value = '登录失败：' + e.message
  }
}
</script>
<style scoped>
.err { color: red; font-size: 12px; }
</style>

3.3 项目列表页 Projects.vue（K15-3）

<template>
  <div>
    <h2>项目列表</h2>
    <input v-model="keyword" placeholder="搜索" />
    <p v-if="loading">加载中...</p>
    <p v-else-if="error" class="err">{{ error }}</p>
    <p v-else-if="!filtered.length">没有匹配的项目</p>
    <ul v-else>
      <li v-for="p in filtered" :key="p.id" @click="goTasks(p.id)">
        {{ p.title }} - {{ p.status }}
      </li>
    </ul>
  </div>
</template>
<script setup>
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { getProjects } from '../api/project'
const router = useRouter()
const keyword = ref('')
const projects = ref([])
const loading = ref(true)
const error = ref('')
const filtered = computed(() => projects.value.filter(p => p.title.includes(keyword.value)))
const goTasks = (id) => router.push(`/projects/${id}/tasks`)
onMounted(async () => {
  try {
    projects.value = await getProjects()
  } catch (e) {
    error.value = e.message || '项目加载失败'
  } finally {
    loading.value = false
  }
})
</script>

本课先用组件本地的 projects 完成页面和状态处理，便于理解页面自身的数据流。第 16 课会把同一份数据迁移到 projectStore，因此迁移时要删除本组件的 projects/loading/error，改用 storeToRefs(projectStore)，不要同时维护两份项目列表。

3.4 任务详情页 TaskDetail.vue（K15-3，动态路由 + props）

<template>
  <div>
    <p v-if="loading">加载中...</p>
    <p v-else-if="error" class="err">{{ error }}</p>
    <template v-else-if="task.id">
      <h2>{{ task.title }}</h2>
      <p>状态：{{ task.status }}</p>
      <p>截止：{{ task.deadline }}</p>
      <button @click="goSubmit">提交成果</button>
    </template>
    <p v-else>任务不存在</p>
  </div>
</template>
<script setup>
import { ref, watch } from 'vue'
import { useRouter } from 'vue-router'
import { getTaskById } from '../api/task'
const props = defineProps(['id'])
const router = useRouter()
const task = ref({})
const loading = ref(true)
const error = ref('')
const goSubmit = () => router.push(`/tasks/${props.id}/submit`)
const loadTask = async () => {
  loading.value = true
  error.value = ''
  task.value = {}
  try {
    task.value = (await getTaskById(props.id)) || {}
  } catch (e) {
    error.value = e.message || '任务加载失败'
  } finally {
    loading.value = false
  }
}
watch(() => props.id, loadTask, { immediate: true })
</script>

3.5 成果提交页 SubmitForm.vue（K15-4）

<template>
  <form @submit.prevent="onSubmit">
    <textarea v-model.trim="form.content" placeholder="成果内容"></textarea>
    <p v-if="error" class="err">{{ error }}</p>
    <button type="submit" :disabled="submitting">{{ submitting ? '提交中...' : '提交' }}</button>
  </form>
</template>
<script setup>
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { submitTask } from '../api/task'
const props = defineProps(['id'])
const form = reactive({ content: '' })
const router = useRouter()
const submitting = ref(false)
const error = ref('')
const onSubmit = async () => {
  if (!form.content) {
    error.value = '请填写成果内容'
    return
  }
  submitting.value = true
  error.value = ''
  try {
    await submitTask(props.id, form)
    alert('提交成功')
    router.push(`/tasks/${props.id}`)
  } catch (e) {
    error.value = e.message || '提交失败'
  } finally {
    submitting.value = false
  }
}
</script>

3.6 评审页 ReviewDetail.vue（K15-4）

<template>
    <form @submit.prevent="onReview">
    <h2>评审</h2>
      <p v-if="loading">加载中...</p>
      <template v-else-if="error">
        <p class="err">{{ error }}</p>
      </template>
      <template v-else-if="notFound">
        <p class="err">评审记录不存在，无法提交评审。</p>
      </template>
      <template v-else>
        <p>提交内容：{{ submission.content || '暂无内容' }}</p>
        <input v-model.number="score" type="number" min="0" max="100" placeholder="分数（0-100）" />
        <textarea v-model.trim="comment" placeholder="评语"></textarea>
        <button type="submit" :disabled="submitting">{{ submitting ? '提交中...' : '提交评审' }}</button>
      </template>
    </form>
</template>
<script setup>
import { ref, watch } from 'vue'
import { getSubmission, reviewSubmission } from '../api/review'
const props = defineProps(['id'])
const submission = ref({})
const score = ref(0)
const comment = ref('')
const loading = ref(true)
const submitting = ref(false)
const error = ref('')
const notFound = ref(false)
const loadSubmission = async () => {
  loading.value = true
  error.value = ''
  notFound.value = false
  submission.value = {}
  score.value = 0
  comment.value = ''
  try {
    const data = await getSubmission(props.id)
    if (!data) {
      notFound.value = true
      return
    }
    submission.value = data
  } catch (e) {
    error.value = e.message || '提交记录加载失败'
  } finally {
    loading.value = false
  }
}
watch(() => props.id, loadSubmission, { immediate: true })
const onReview = async () => {
  if (score.value < 0 || score.value > 100 || !Number.isFinite(score.value)) {
    error.value = '分数必须是 0～100 的数字'
    return
  }
  if (!comment.value) {
    error.value = '请填写评语'
    return
  }
  submitting.value = true
  error.value = ''
  try {
    await reviewSubmission(props.id, { score: score.value, comment: comment.value })
    alert('评审完成')
  } catch (e) {
    error.value = e.message || '评审失败'
  } finally {
    submitting.value = false
  }
}
</script>

src/stores/user.js（Mock 和真实接口都通过 src/api/user.js 调用）：

import { defineStore } from 'pinia'
import { login as loginApi } from '../api/user'

const readUserInfo = () => {
  try { return JSON.parse(localStorage.getItem('userInfo') || 'null') }
  catch { return null }
}

export const useUserStore = defineStore('user', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    userInfo: readUserInfo()
  }),
  getters: {
    isLogin: (state) => Boolean(state.token),
    username: (state) => state.userInfo?.username || '未登录',
    role: (state) => state.userInfo?.role || 'guest'
  },
  actions: {
    async login({ username, password, role }) {
      if (!username || !password || password.length < 6) throw new Error('用户名或密码不正确')
      const data = await loginApi({ username, password, role: role || 'student' })
      if (!data?.token || !data?.userInfo) throw new Error('登录响应缺少 token 或 userInfo')
      this.token = data.token
      this.userInfo = data.userInfo
      localStorage.setItem('token', this.token)
      localStorage.setItem('userInfo', JSON.stringify(this.userInfo))
      return data
    },
    logout() {
      this.token = ''
      this.userInfo = null
      localStorage.removeItem('token')
      localStorage.removeItem('userInfo')
    }
  }
})

src/utils/request.js（与第 14 课的响应契约一致）：

import axios from 'axios'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000,
  headers: { 'Content-Type': 'application/json' }
})

request.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) config.headers.Authorization = 'Bearer ' + token
  return config
})

request.interceptors.response.use(
  (response) => {
    const body = response.data
    if (body && typeof body === 'object' && 'code' in body) {
      if (body.code !== 200) throw new Error(body.message || '业务请求失败')
      return body.data
    }
    return body
  },
  (error) => Promise.reject(error)
)

export default request

  document.title = to.meta.title || 'CTTS-Frontend'
  const userStore = useUserStore()
  if (to.meta.requireAuth && !userStore.isLogin) {

  }

})

<script setup>
import { ref, watch } from 'vue'
import { useRouter } from 'vue-router'
import { getTasks } from '../api/task'
const props = defineProps(['id'])
const router = useRouter()
const tasks = ref([])
const loading = ref(true)
const error = ref('')
const loadTasks = async () => {
  loading.value = true
  error.value = ''
  try { tasks.value = await getTasks(props.id) }
  catch (e) { error.value = e.message || '任务加载失败' }
  finally { loading.value = false }
}
watch(() => props.id, loadTasks, { immediate: true })
</script>
<template>
  <section>
    <h2>任务列表</h2>
    <p v-if="loading">加载中...</p>
    <p v-else-if="error">{{ error }}</p>
    <p v-else-if="!tasks.length">暂无任务</p>
    <ul v-else>
      <li v-for="task in tasks" :key="task.id">
        {{ task.title }} <button @click="router.push('/tasks/' + task.id)">查看</button>
      </li>
    </ul>
  </section>
</template>
